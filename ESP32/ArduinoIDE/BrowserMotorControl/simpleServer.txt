#include <WiFi.h>

const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";

WiFiServer server(80);

void setup()
{
  Serial.begin(115200);

  WiFi.begin(ssid, password);

  Serial.print("Connecting");

  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi Connected!");
  Serial.print("ESP32 IP Address: ");
  Serial.println(WiFi.localIP());

  server.begin();
  Serial.println("HTTP Server Started");
}

void loop()
{
  WiFiClient client = server.available();//checks for new connections

  if (client)//if a new client is connected
  {
    Serial.println("Client Connected");

    // Wait until the browser sends a request
    // if client is still connected but hasn't send any requests yet, wait till it sends a request.
    // client.connected(): checks if client is still connected or not
    // server.available(): only checks for new connectons, doesn't give connection status of current client.
    // client.available(): checks if client has send any data/requests.
    //                     if data is waiting in the buffer to read, it returns a true condition 
    while (client.connected() && !client.available())
    {
      delay(1);
    }

    // Read and print the HTTP request
    while (client.available())
    {
      String request = client.readStringUntil('\n');
      Serial.println(request);
	
	//last line of an HTTP request is a blank line(contains only \r\n), request==that line means its over
      if (request == "\r")
        break;   // End of HTTP headers
    }

    // Send HTTP Response
    client.println("HTTP/1.1 200 OK");
    client.println("Content-Type: text/plain");
    client.println("Connection: close");
    client.println();

    client.println("Hello from ESP32!");

    delay(1);
    client.stop();
